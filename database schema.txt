CREATE TABLE documentHolder (
    documentHolder_id INTEGER PRIMARY KEY AUTOINCREMENT,
    reference_no TEXT NOT NULL,
    title TEXT NOT NULL,
    department_id INTEGER DEFAULT 1,
    current_version_id TEXT DEFAULT NULL,
    status TEXT DEFAULT NULL,
    created_by INTEGER DEFAULT 1,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE documentHolder_Movement (
    movement_id INTEGER PRIMARY KEY AUTOINCREMENT,
    documentHolder_id INTEGER,
    from_department_id INTEGER,
    to_department_id INTEGER,
    movement_type TEXT DEFAULT 'Transfer',
    approved_by INTEGER,
    moved_by INTEGER,
    moved_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    remarks TEXT DEFAULT NULL
);

CREATE TABLE documentHolder_version (
    version_id INTEGER PRIMARY KEY AUTOINCREMENT,
    documentHolder_id INTEGER,
    version_number TEXT,
    file_path TEXT,
    file_hash TEXT,
    file_size TEXT,
    update_by INTEGER,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    status TEXT DEFAULT 'Active'
);

CREATE TABLE tblModuleData (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    Name TEXT NOT NULL,
    Code TEXT NOT NULL,
    TypeID INTEGER NOT NULL,
    ModuleType TEXT,
    ParentID INTEGER DEFAULT 1,
    isactive INTEGER,
    createdby INTEGER DEFAULT 1,
    createdon DATETIME DEFAULT CURRENT_TIMESTAMP,
    updatedby INTEGER,
    updatedon DATETIME,
    Discriminator TEXT
);

CREATE TABLE tblModules (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    ModuleID INTEGER DEFAULT 0,
    ModuleName TEXT NOT NULL,
    Viewreadonly INTEGER
);

CREATE TABLE tblProfile (
    ID INTEGER PRIMARY KEY AUTOINCREMENT,
    profilename TEXT NOT NULL,
    uid INTEGER NOT NULL,
    isactive INTEGER DEFAULT 1,
    isdeleted INTEGER DEFAULT 0,
    createdby INTEGER DEFAULT 1,
    createdon DATETIME DEFAULT CURRENT_TIMESTAMP,
    updatedby INTEGER,
    updatedon DATETIME,
    companyid INTEGER DEFAULT 1,
    Roleid INTEGER DEFAULT 1
);

CREATE TABLE tblProfileDetails (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    profileid INTEGER NOT NULL,
    moduleId INTEGER NOT NULL,
    CanInsert INTEGER,
    CanUpdate INTEGER,
    CanDelete INTEGER,
    CanView INTEGER,
    isactive INTEGER,
    isdeleted INTEGER,
    createdby INTEGER,
    createdon DATETIME,
    updatedby INTEGER,
    updatedon DATETIME
);

CREATE TABLE tblReaderlog (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    code TEXT,
    logDatetime DATETIME DEFAULT CURRENT_TIMESTAMP,
    logStatus TEXT,
    LogMessage TEXT DEFAULT NULL
);

PRAGMA foreign_keys = ON;

CREATE TABLE documentHolder_Movement (
    movement_id INTEGER PRIMARY KEY AUTOINCREMENT,

    documentHolder_id INTEGER,
    from_department_id INTEGER,
    to_department_id INTEGER,

    movement_type TEXT DEFAULT 'Transfer',

    approved_by INTEGER,
    moved_by INTEGER,

    moved_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    remarks TEXT,

    FOREIGN KEY (documentHolder_id)
        REFERENCES documentHolder(documentHolder_id)
        ON DELETE CASCADE
);

CREATE TABLE documentHolder_version (
    version_id INTEGER PRIMARY KEY AUTOINCREMENT,

    documentHolder_id INTEGER NOT NULL,
    version_number INTEGER NOT NULL,

    file_path TEXT NOT NULL,
    file_hash TEXT,
    file_size INTEGER,

    update_by INTEGER,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    status TEXT DEFAULT 'Active',

    FOREIGN KEY (documentHolder_id)
        REFERENCES documentHolder(documentHolder_id)
        ON DELETE CASCADE
);

CREATE INDEX idx_document_department
ON documentHolder(department_id);

CREATE INDEX idx_version_document
ON documentHolder_version(documentHolder_id);

CREATE INDEX idx_movement_document
ON documentHolder_Movement(documentHolder_id);

CREATE INDEX idx_movement_department
ON documentHolder_Movement(to_department_id);

CREATE UNIQUE INDEX idx_version_unique
ON documentHolder_version(documentHolder_id, version_number);

SELECT
d.documentHolder_id,
d.title,    
v.version_number,
v.file_path
FROM documentHolder d
JOIN documentHolder_version v
ON d.current_version_id = v.version_id;
